rules_version = '2';

// QREN Esports — target security rules.
// IMPORTANT: Ye rules tab apply karo jab code refactor ho jaye:
// settings/public, settings/private, tournament_private, referralCodes,
// referrals, joinLocks aur admin login. Current code in collections ko
// abhi use nahi karta. /admins/{uid} doc console se banega.

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function uid() {
      return request.auth.uid;
    }

    function isOwner(userId) {
      return signedIn() && uid() == userId;
    }

    function isAdmin() {
      return signedIn() && (
        uid() == 'E8mVnZXrGhQVvVFr5Sj3tdLZrSg1'
        || exists(/databases/$(database)/documents/admins/$(uid()))
      );
    }

    function changedOnly(keys) {
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly(keys);
    }

    function tournamentDoc(id) {
      return /databases/$(database)/documents/tournaments/$(id);
    }

    function joinDoc(tid) {
      return /databases/$(database)/documents/joins/$(tid + '_' + uid());
    }

    function lockDoc(tid) {
      return /databases/$(database)/documents/joinLocks/$(tid + '_' + uid());
    }

    // Client se admin doc read/write nahi hoga. Console/Admin SDK se banao.
    match /admins/{adminId} {
      allow get: if isOwner(adminId);
      allow list, write: if false;
    }

    // Player apna profile read kar sakta hai; wallet client se change nahi hoga.
    match /users/{userId} {
      allow get: if isAdmin() || isOwner(userId);
      allow list: if isAdmin();
      allow create: if isOwner(userId)
        && request.resource.data.wallet.balance == 0
        && request.resource.data.wallet.totalEarnings == 0
        && request.resource.data.wallet.totalDeposits == 0
        && request.resource.data.wallet.totalWithdrawals == 0;
      allow update: if isOwner(userId) && changedOnly(['userName', 'photo', 'userId', 'email', 'emailVerified', 'emailVerificationRequestedAt'])
        && (
          !request.resource.data.diff(resource.data).affectedKeys().hasAny(['photo'])
          || request.resource.data.photo == ''
          || (request.resource.data.photo is string
              && request.resource.data.photo.matches('^data:image/(jpeg|png|webp);base64,')
              && request.resource.data.photo.size() <= 250000)
        )
        && (
          !request.resource.data.diff(resource.data).affectedKeys().hasAny(['userId'])
          || (request.resource.data.userId.matches('^[0-9]{5}$') && !('userId' in resource.data))
        )
        && (
          !request.resource.data.diff(resource.data).affectedKeys().hasAny(['email', 'emailVerified', 'emailVerificationRequestedAt'])
          || (request.resource.data.email is string
              && request.resource.data.email.size() <= 120
              && request.resource.data.email.matches('^[^@ ]+@[^@ ]+[.][^@ ]+$')
              && request.resource.data.emailVerified == false
              && request.resource.data.emailVerificationRequestedAt == request.time)
        );
      allow update: if isAdmin()
        && changedOnly(['emailVerified'])
        && request.resource.data.emailVerified == true
        && resource.data.email is string
        && resource.data.email.size() > 0;
      allow delete: if false;
    }

    // 5-digit public user ID reservation. Har ID sirf ek hi account le sakta hai.
    match /userIds/{userIdCode} {
      allow get: if signedIn();
      allow list: if isAdmin();
      allow create: if signedIn()
        && request.resource.data.uid == uid()
        && request.resource.data.userId == userIdCode
        && userIdCode.matches('^[0-9]{5}$');
      allow update, delete: if isAdmin();
    }

    // Public referral code lookup. Is doc me sirf code, uid, userName rakho.
    match /referralCodes/{code} {
      allow get: if true;
      allow list: if false;
      allow create: if signedIn()
        && request.resource.data.uid == uid()
        && request.resource.data.code == code;
      allow update, delete: if isAdmin();
    }

    // Referral dashboard ke liye owner-only list. Query: referrerUid == uid.
    match /referrals/{referralId} {
      allow get, list: if isAdmin() || (signedIn() && resource.data.referrerUid == uid());
      allow create: if signedIn()
        && request.resource.data.referredUid == uid()
        && referralId == request.resource.data.referrerUid + '_' + uid();
      allow update, delete: if isAdmin();
    }

    // Public tournament doc me roomId/roomPassword kabhi mat rakho.
    match /tournaments/{id} {
      allow read: if true;
      allow create, delete: if isAdmin();
      allow update: if isAdmin() || (
        signedIn()
        && changedOnly(['joinedCount', 'status'])
        && request.resource.data.joinedCount == resource.data.joinedCount + 1
        && (resource.data.totalSlots == 0 || request.resource.data.joinedCount <= resource.data.totalSlots)
        && request.resource.data.status in ['open', 'upcoming', 'full']
        && !exists(lockDoc(id))
        && existsAfter(lockDoc(id))
        && getAfter(lockDoc(id)).data.counted == true
      );
    }

    // Room credentials sirf joined player aur admin padh sakte hain.
    match /tournament_private/{id} {
      allow get: if isAdmin() || (signedIn() && exists(joinDoc(id)));
      allow list: if isAdmin();
      allow write: if isAdmin();
    }

    match /joins/{joinId} {
      allow get, list: if isAdmin() || (signedIn() && resource.data.userId == uid());
      allow create: if signedIn()
        && request.resource.data.userId == uid()
        && joinId == request.resource.data.tournamentId + '_' + uid()
        && request.resource.data.status == 'joined'
        && get(tournamentDoc(request.resource.data.tournamentId)).data.status in ['open', 'upcoming']
        && (get(tournamentDoc(request.resource.data.tournamentId)).data.totalSlots == 0
            || get(tournamentDoc(request.resource.data.tournamentId)).data.joinedCount
               < get(tournamentDoc(request.resource.data.tournamentId)).data.totalSlots)
        && !exists(lockDoc(request.resource.data.tournamentId));
      allow update, delete: if isAdmin();
    }

    // Ek player ek tournament me sirf ek baar count badha sakta hai.
    match /joinLocks/{lockId} {
      allow get: if isAdmin() || (signedIn() && resource.data.userId == uid());
      allow list: if isAdmin();
      allow create: if signedIn()
        && request.resource.data.userId == uid()
        && lockId == request.resource.data.tournamentId + '_' + uid()
        && request.resource.data.counted == true
        && existsAfter(joinDoc(request.resource.data.tournamentId));
      allow update: if false;
      allow delete: if isAdmin();
    }

    match /results/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Transactions/wallet sirf admin ya backend likhega.
    match /transactions/{id} {
      allow get, list: if isAdmin() || (signedIn() && resource.data.userId == uid());
      allow write: if isAdmin();
    }

    // Legacy players data ko public/private me split karna hai.
    match /players_public/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /players_private/{id} {
      allow read, write: if isAdmin();
    }

    match /settings/public {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /settings/private {
      allow read, write: if isAdmin();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
